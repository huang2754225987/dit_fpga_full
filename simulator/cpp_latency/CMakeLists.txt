cmake_minimum_required(VERSION 3.16)
project(supra2_latency_model LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(supra2_latency_model src/main.cpp)
target_include_directories(supra2_latency_model PRIVATE include)

add_library(ddr4_device src/ddr4_device.cpp)
target_include_directories(ddr4_device PUBLIC include)
add_library(mig_axi_core src/mig_axi_model.cpp)
target_include_directories(mig_axi_core PUBLIC include)
target_link_libraries(mig_axi_core PUBLIC ddr4_device)
add_library(npu_pipeline_core src/npu_pipeline_model.cpp)
target_include_directories(npu_pipeline_core PUBLIC include)
target_link_libraries(npu_pipeline_core PUBLIC mig_axi_core)
add_executable(ddr4_replay src/ddr4_replay.cpp)
target_link_libraries(ddr4_replay PRIVATE ddr4_device)
add_executable(mig_axi_model src/mig_axi_main.cpp)
target_link_libraries(mig_axi_model PRIVATE mig_axi_core)
add_executable(npu_pipeline_demo src/npu_pipeline_demo.cpp)
target_link_libraries(npu_pipeline_demo PRIVATE npu_pipeline_core)
add_executable(model_characterization src/model_characterization.cpp)
target_link_libraries(model_characterization PRIVATE npu_pipeline_core)
add_executable(attention_overlap_model src/attention_overlap_model.cpp)
add_executable(dma_width_model src/dma_width_model.cpp)
add_executable(gemm_architecture_model src/gemm_architecture_model.cpp)

include(CTest)
if(BUILD_TESTING)
  add_test(NAME gemm_architecture_tests COMMAND gemm_architecture_model --self-test)
  add_test(NAME dma_width_tests COMMAND dma_width_model --self-test)
  add_test(NAME attention_overlap_tests COMMAND attention_overlap_model --self-test)
  add_executable(npu_pipeline_tests tests/npu_pipeline_tests.cpp)
  target_link_libraries(npu_pipeline_tests PRIVATE npu_pipeline_core)
  add_test(NAME npu_pipeline_tests COMMAND npu_pipeline_tests)
  add_executable(latency_model_tests tests/latency_model_tests.cpp)
  target_include_directories(latency_model_tests PRIVATE include)
  add_test(NAME latency_model_tests COMMAND latency_model_tests)
  add_executable(ddr4_device_tests tests/ddr4_device_tests.cpp)
  target_link_libraries(ddr4_device_tests PRIVATE ddr4_device)
  add_test(NAME ddr4_device_tests COMMAND ddr4_device_tests)
endif()
